\makeatletter
\renewcommand{\maketitle}{
  \begin{titlepage}
    \centering
    \vspace{2cm}
    {\Huge\bfseries \@title\par}
    \vspace{3cm}
    {\Large \@author\par}
    \vfill
    {\large \@date\par}
  \end{titlepage}
  \setcounter{page}{1}
}
\makeatother

% ======================
% URL and hyperref configuration
% ======================
\usepackage[hyphens]{url}       % Basic URL handling with hyphen breaks
\usepackage{xurl}               % Allows breaks at ANY character (must come before hyperref)
\usepackage[colorlinks=true,
            breaklinks=true,
            hypertexnames=false,
            allcolors=blue]{hyperref}  % Load ONCE with all options
\urlstyle{same}                % Maintains document font style for URLs

% ======================
% Float and text handling
% ======================
\usepackage{placeins}          % Adds \FloatBarrier
\usepackage{afterpage}         % Better page break control
\usepackage{parskip}           % Better paragraph spacing
\setlength{\emergencystretch}{3em} % Prevent overfull hboxes

% ======================
% Table configuration
% ======================
\usepackage{array}
\usepackage{booktabs}
\usepackage{tabularx}
\usepackage{adjustbox}
\usepackage{ragged2e}
\usepackage{multirow}
\usepackage{siunitx} 

% Caption setup
\usepackage[justification=raggedright,singlelinecheck=false]{caption}
\captionsetup[table]{position=below} 
\usepackage{float}
\floatstyle{plaintop}
\restylefloat{table}

\makeatletter
\@ifpackageloaded{caption}{%
  \DeclareCaptionFormat{myformat}{\raggedright#1#2#3}
  \captionsetup[table]{format=myformat,position=below}
}{}
\makeatother

% Custom column types
\newcolumntype{L}{>{\RaggedRight\arraybackslash}X} % Left-aligned
\newcolumntype{R}{>{\RaggedLeft\arraybackslash}X}  % Right-aligned
\newcolumntype{C}{>{\Centering\arraybackslash}X}   % Centered

% Font size control
\usepackage{etoolbox}
\captionsetup{font=footnotesize}
\renewcommand{\captionlabelfont}{\footnotesize}
\renewcommand{\captionfont}{\footnotesize}

% Table font size
\AtBeginEnvironment{tabular}{\footnotesize}
\AtBeginEnvironment{longtable}{\footnotesize}
\AtBeginEnvironment{tabularx}{\footnotesize}

% ======================
% TOC handling
% ======================
\let\oldtoc\tableofcontents
\renewcommand{\tableofcontents}{%
  \clearpage
  \pagestyle{empty}
  \oldtoc
  \clearpage
  \pagestyle{plain}
}
